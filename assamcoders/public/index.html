Google Sheets Live Sync Configuration

Collect every builder submission directly as rows in your own Google Sheet.

Saved automatically in browser storage. Every card created will append a row immediately.

How to setup in 30 seconds:
  1. Create a new Google Sheet.
  2. Click Extensions → Apps Script.
  3. Delete existing code, paste the script below, and click Deploy → New deployment.
  4. Select type Web App, set Who has access to Anyone, click Deploy, and paste the URL above!
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      new Date(),
      data.name || '',
      data.role || '',
      data.idea || '',
      data.tagline || '',
      data.problem || '',
      data.website || '',
      data.phone || '',
      data.theme || '',
      data.shareUrl || ''
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
